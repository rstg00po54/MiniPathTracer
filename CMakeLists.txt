cmake_minimum_required(VERSION 3.16)
project(MiniPathTracer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# std::thread requires an explicit pthread dependency on older Linux/glibc.
# FindThreads also handles the correct platform flags on macOS and Windows.
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)

add_library(mpt_core src/pathtracer.cpp)
target_include_directories(mpt_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(mpt_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(mpt_core PRIVATE /W4)
else()
  target_compile_options(mpt_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(MiniPathTracer src/main.cpp)
target_link_libraries(MiniPathTracer PRIVATE mpt_core)

option(MPT_ENABLE_SDL2 "Enable SDL2 live preview if available" ON)
if(MPT_ENABLE_SDL2)
  find_package(SDL2 QUIET CONFIG)
  if(NOT SDL2_FOUND)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
    endif()
  endif()
  if(TARGET SDL2::SDL2)
    target_link_libraries(MiniPathTracer PRIVATE SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(MiniPathTracer PRIVATE SDL2::SDL2main)
    endif()
    target_compile_definitions(MiniPathTracer PRIVATE MPT_WITH_SDL2=1)
    message(STATUS "SDL2 preview enabled (CMake target)")
  elseif(TARGET PkgConfig::SDL2)
    target_link_libraries(MiniPathTracer PRIVATE PkgConfig::SDL2)
    target_compile_definitions(MiniPathTracer PRIVATE MPT_WITH_SDL2=1)
    message(STATUS "SDL2 preview enabled (pkg-config)")
  else()
    message(STATUS "SDL2 not found: building headless renderer (PPM/BMP output)")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mpt_tests tests/test_pathtracer.cpp)
  target_link_libraries(mpt_tests PRIVATE mpt_core)
  add_test(NAME mpt_core_tests COMMAND mpt_tests)
endif()
